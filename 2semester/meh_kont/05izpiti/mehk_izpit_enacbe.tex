  \subsection{Splošne deformacije}

  Vektor premika \( \mathbf{u} (\mathbf{r}) = \mathbf{r}' - \mathbf{r} \) in deformacijski tenzor

  \( u_{ik} = \frac{1}{2} \left( \frac{\partial u_{i}}{\partial x_{k}} + \frac{\partial u_{k}}{\partial x_{i}}  \right) \).

  \( i \) pove komponento, \( k \) pove stolpec v tenzorju!

  \textbf{Sferične koordinate} 

  \( u_{rr} = \partial_r u_r \)

  \( u_{\theta \theta} = \frac{1}{r} \partial_{\theta} u_{\theta} + \frac{u_r}{r} \)

  \( u_{\phi\phi} = \frac{1}{\sin \theta} \frac{1}{r} \partial_{\phi} u_{\phi} + \frac{u_r}{r} + \frac{u_{\theta}}{r \tan \theta} \)

  \( u_{r\theta} = \frac{1}{2} \left( \partial_r u_{\theta} - \frac{u_{\theta}}{r} + \frac{1}{r} \partial_{\theta} u_r \right) \)

  \( u_{r\phi} = \frac{1}{2} \left( \partial_r u_{\phi} - \frac{u_{\phi}}{r} + \frac{1}{r \sin \theta} \partial_{\phi} u_r \right) \)

  \( u_{\theta \phi} = \frac{1}{2} \left( \frac{1}{r} \partial_{\theta} u_{\phi} - \frac{u_{\phi}}{r \tan \theta} + \frac{1}{r \sin \theta} \partial_{\phi} u_{\theta} \right) \)

  \textbf{Cilindrične koordinate}

  \( u_{rr} = \partial_r u_r \)

  \( u_{\phi\phi} = \frac{1}{r} \partial_{\phi} u_{\phi} + \frac{u_r}{r} \)

  \( u_{zz} = \partial_z u_z \)

  \( u_{r\phi} = \frac{1}{2} \left( \partial_r u_{\phi} - \frac{u_{\phi}}{r} + \frac{1}{r} \partial_{\phi} u_r \right) \)

  \( u_{rz} = \frac{1}{2} \left( \partial_z u_r + \partial_r u_z \right) \)

  \( u_{\phi z} = \frac{1}{2} \left( \frac{1}{r} \partial_{\phi} u_z + \partial_z u_{\phi} \right) \)

  \textbf{Hookov zakon}

  \( p_{ij} = 2\mu u_{ij} + \lambda u_{kk} \delta_{ij} = \frac{E}{1 + \sigma} \left( u_{ij} + \frac{\sigma}{1 - 2\sigma} u_{kk} \delta_{ij} \right) \),

  kjer je \( u_{kk} = \tr \mathsf{u} \), \( \mu = \frac{E}{2 (1 + \sigma}, \lambda = \frac{E \sigma}{(1 - 2 \sigma) (1 + \sigma)} \) Laméjeva el. koeficienta, \( E = \frac{p_{zz}}{u_{zz}} = \frac{\mu \left( 2 \mu + 3 \lambda \right)}{\mu + \lambda} \) Youngov modul in \( \sigma = - \frac{u_{xx/yy}}{u_{zz}} = \frac{1}{2(1 + \frac{\mu}{\lambda})} \) Poissonovo število. Obratno

  \( u_{ij} = \frac{1}{2\mu} p_{ij} - \frac{\lambda}{2 \mu \left( 2\mu + 3 \lambda \right)} p_{kk} \delta_{ij} = \frac{1}{E} \left[ \left( 1+ \sigma \right) p_{ij} - \sigma p_{kk} \delta_{ij} \right] \)

  \textbf{Cauchyjeva enačba v ravnovesju}

  \( \ddot{u}_i = f_i + \frac{\partial }{\partial x_{k}} p_{ik} = 0 \), kjer so \( f_i \) volumnsko porazdeljene zunanje sile. 

  Za površinske sile velja \( p_{ik}  n_k = F_i \), kjer je \( n_k \) vektor normale na površino.

  \textbf{Navierova enačba}

  V indeksni obliki

  \( \rho \ddot{u}_i = f_i + \mu \nabla ^2 u_i + \left( \mu + \lambda \right) \frac{\partial }{\partial x_{i}} \left( \frac{\partial }{\partial x_{k}} u_k \right) \)

  in v vektorski obliki

  \( \rho \ddot{\mathbf{u}} = \mathbf{f} + \mu \nabla ^2 \mathbf{u} + \left( \mu + \lambda \right) \nabla \nabla \cdot \mathbf{u} \)

  z \( E, \sigma \)

  \( \rho \ddot{\mathbf{u}} = \mathbf{f} + \frac{E}{2(1 + \sigma)} \left( \nabla ^2 \mathbf{u} + \frac{1}{1 - 2\sigma} \nabla \nabla \cdot \mathbf{u} \right) \)

  \begin{primer}[Enoosna obremenitev]
    Položen valj, katerega daljša stranica leži v smeri \( z \). Površinski sili na osnovni ploskvi \( S \), ki sta vzporedni ravnini \( xy \) zapišemo kot \( p_{ij} (0, 0, 1)^T = (0, 0, - F/S)  \)
  \end{primer}

  \begin{primer}
    Hidrostatičen tlak iz predavanj predstavimo z \( p_{ik} = - p \delta_{ik} \)
  \end{primer}
  \begin{primer}
    Strižna obremenitev ima dve izvendiagonalni komponenti neničelni \( p_{ik} = p (\delta_{iz} \delta_{kx} + \delta_{ix} \delta_{kz} \)
  \end{primer}

  Ob uporabi identitete \( \nabla \nabla \cdot = \nabla ^2 + \nabla \times \nabla \times \), kjer je pogosto drugi člen ničeln, je stacionarna Navierova enačba 

  \( \mathbf{f} + \left( 1 + \frac{1}{1 - 2 \sigma} \right) \frac{E}{2 (1 + \sigma)} \nabla \nabla \cdot \mathbf{u} - \frac{E}{2(1 + \sigma)} \nabla \times \nabla \times \mathbf{u} = 0  \).

  \begin{primer}
    Drugi člen je ničeln za \( \mathbf{u} = u_r (r) \mathbf{e}_i \), kjer je \( i = r, \phi \)
  \end{primer}
  
  Sprememba volumna je

  \( \frac{\Delta V}{V} = u_{kk} = \frac{1}{2 \mu + 3 \lambda} p_{kk} \), kjer velja za sled tudi \( u_{kk} = \nabla \cdot \mathbf{u} \).

  \begin{primer}
    V primeru deformacijskega vektorja oblike \( \mathbf{u} = u_r (r) \mathbf{e}_r + u_{\phi} (r) \mathbf{e}_{\phi} \) ločimo na \( \mathbf{u} = \mathbf{u}_1 + \mathbf{u}_2\), kjer je \( \nabla \cdot u_2 = 0 \) in \( \nabla \times u_1 = 0 \).
  \end{primer}

  \subsection{Kristali}

  Elastična prosta energija je

  \( f(\mathsf{u}) = f_0 + \frac{1}{2} C_{ijkl} u_{ij} u_{kl} \),

  kjer je \( C_{ijkl} \) tenzor el. konstant z največ 21 neodvisnimi konstantami!

  \( p_{ij} = \frac{\partial }{\partial u_{ij}} f = C_{ijkl} u_{kl} \).
  
  \begin{primer}[Zrcaljenje tetragonalnega kristala]
    Zrcaljenje \( x \to -x \) (preko ravnine \( zy \)) po definiciji

    \( u_{xy} = \frac{1}{2} \left( \partial_y u_x + \partial_x u_y \right) \to \frac{1}{2} \left( \partial_y (- u_x ) + \partial_{(-x)} u_y \right) = - u_{yx} \)

    Elastična prosta energija se pri zrcaljenju ne spremeni, torej bo zrcaljenje \( C_{xyxx} u_{xy} u_{xx} \to C_{xyxx} (-u_{xy}) u_{xx} \) veljalo za \( C_{xyxx} = 0 \).
  \end{primer}
  
  \section{Plošče}

  Odmik \( u_z = \zeta (x, y) \) opisuje enačba ravnovesja

  \( D ^2 \Delta ^2_{\perp} u_z - P(x, y) = 0, \quad D = K = \frac{E h ^3}{12 (1 - \sigma ^2)} \),

  kjer je \( P(x, y) \) površinska gostota sil (tlak). 

  \begin{primer}
    \( \Delta_{\perp} ^2 f = \Delta_{\perp} \left[ \frac{1}{r} \partial_r \left( r \partial_r f \right) \right] = \frac{1}{r} \partial_r \left\{ r \partial_r \left[ \frac{1}{r} \partial_r \left( r \partial_r f \right) \right] \right\} \)
  \end{primer}
  
  \textbf{Togo vpeta plošča}

  ima robna pogoja

  \( \left. u_z \right|_{\partial S} = 0 \quad \text{ in } \quad \left. \frac{\partial }{\partial \mathbf{n}} u_z \right|_{\partial S} = 0\),

  kjer je \( \partial S \) rob plošče in \( \partial_{\mathbf{n}} \) odvod v smeri, ki je vzporedna z ravnino plošče in pravokotna na njen rob. 

  \textbf{Prislonjena plošča}

  ima prvi robni pogoj isti, drugi je

  \( \left. \left( \frac{\partial ^2 u_{z}}{\partial \mathbf{n} ^2} + \sigma \frac{\partial \phi}{\partial l}  \frac{\partial u_{z}}{\partial \mathbf{n}}  \right)\right|_{\partial S} = 0 \),

  kjer je \( \phi \) kot med tangento na rob plošče in vodoravno koordinato. Ponavadi je \( \partial_l \phi = \frac{1}{R} \).

  \begin{primer}
    Na okroglo ploščo polmera \( R \) pritiska ploskovna gostota s silo \( F_0 \) v koncentrične krogu polmera \( r_0 < R \). Ploskovna gostota je potem \( F_0 = \int\limits_{}^{} P \, \mathrm{d} S \).

    \( P = \frac{F_0 \delta(r - r_0)}{2 \pi r} \)
  \end{primer}

  Funkcional proste energije je
  \vspace{-10pt}
  \begin{align*}
    F = F_0 &+ \frac{1}{2} D \int\limits_{}^{} \left\{ \left( \nabla ^2_{\perp} u \right) ^2 + 2 (1 + \sigma) \cdot \right. \\
    &\cdot \left. \left[ {\left( \frac{\partial ^2 u}{\partial x \partial y}  \right)}^2 - \frac{\partial ^2 u}{\partial x ^2} \frac{\partial ^2 u}{\partial y ^2}  \right] \right\} \, \mathrm{d} S - \int\limits_{}^{} P u_z \, \mathrm{d} S
  \end{align*}

  Velja tudi

  \( u_{xy}  ^2 - u_{xx} u_{yy} = \nabla \cdot \left( u_y u_{xy}, -u_y u_{xx } \right) \), kjer indeksi označujejo \textbf{odvode}.
  \subsection{Palice}
  \subsubsection{Torzija}
  Torzijski zasuk zapišemo kot \( \mathrm{d} \phi = \tau \mathrm{d} z \), kjer je \( \tau \) torzijski zasuk na enoto dolžine in je konst. Majhna torzijska deformacije je toga rotacija \( \mathbf{u} = \bf{\phi} \times \mathbf{r}\), kjer je \( \psi(x, y) \) torzijska funkcija

  \( \mathbf{u} = (- \tau y z, \tau x y, \tau \psi(x, y) \).

  Po definiciji def. tenzorja sta neničelni komponenti

  \( u_{xz} = \frac{\tau}{2} \left( \partial_x \psi - y \right) \) in

  \( u_{yz} = \frac{\tau}{2} \left( \partial_y \psi + x \right) \).

  \textbf{Ohranja se prostornina} \( \tr \mathsf{u} = u_{kk} = 0 \).

  Neničelne komponente tenzorja \( p_{ij} \) so zaradi simetričnosti

  \( p_{xz} = p_{zx} = \mu \tau \left( \partial_x \psi - y \right) = 2 \mu \tau \partial_y \chi \) in

  \( p_{yz} = p_{zy} = \mu \tau \left( \partial_y \psi + x \right) = - 2 \mu \tau \partial_x \chi \)

  Cauchyjeva enačba v ravnovesju poda

  \( f_z = \partial_z p_{zx} + \partial_z p_{zy} = 0 \implies \quad \nabla ^2 \psi(x, y) = 0  \).

  Definiramo prirejeno torzijsko funkcijo \( \chi (x, y) \) z

  \( \partial_x \psi = y + 2 \partial_y \chi \) in

  \( \partial_y \psi = -x  -2 \partial_x \chi \).

  Velja torej \( \nabla ^2 \chi = - 1 \).

  Prosti robni pogoj je \( p_{zx} n_x + p_{zy} n_y = 0 \), za enostavno povezano območje \( \left. \chi \right|_{\text{rob}} = 0 \).

  Navor na palico \( M_z = - 2 \mu \tau \int\limits_{}^{} \mathbf{r} \cdot \nabla \chi \, \mathrm{d} S \approx 4 \mu \tau \int\limits_{}^{} \chi \, \mathrm{d} S \) za en. pov. obm. 

  \begin{primer}[Elipsast valj]
    Enačbo \( \nabla ^2 \chi = -1 \) rešimo z nastavkom

    \( \chi = C \left[ \frac{x ^2}{a ^2} + \frac{y ^2}{b ^2} - 1 \right] \)

    in dobimo \( C = - a ^2 b ^2 / \left( 2 \left[ a ^2 + b ^2 \right] \right) \). Potem je

    \( \chi (x, y) = - \frac{a ^2 b ^2}{2 (a ^2 + b ^2)} \left[ \frac{x ^2}{a ^2} + \frac{y ^2}{b ^2} - 1 \right] \) in

    \( \psi(x, y) = xy \frac{b ^2 - a ^2}{a ^2 + b ^2} \).
  \end{primer}

  \textbf{Globalna teorija palic}

  Tangenta \( \hat{\mathbf{t}} = \partial_l \mathbf{r} (l) \)

  Normala \( \hat{\mathbf{n}} = R \partial_l \hat{\mathbf{t}} (l) \)

  Ravnovesje sil na del palice je \( \frac{\mathrm{d} \mathbf{F}}{\mathrm{d} l} + \mathbf{K} = 0 \), kjer je \( \mathbf{K} \) dolžinska gostota zunanjih sil na palico.

  Ravnovesje navoro je \( \frac{\mathrm{d} \mathbf{M}}{\mathrm{d} l} + \hat{\mathbf{t}} \times \mathbf{F} = 0 \)

  \textbf{Majhen upogib pravkoten na os z}

  \( \mathbf{M} = E \mathsf{I} \left( - \partial_z ^2 y, \partial_z ^2 x \right) \)

  \( \mathbf{F} = - E \mathsf{I} \partial_z ^3 \mathbf{r} + F_z \partial_z \mathbf{r} \) za \( \mathbf{r} = (x, y) \)

  \( E \mathsf{I} \partial_z ^4 \mathbf{r} - F_z \partial_z ^2 x - \partial_z F_z \partial_z x - \mathbf{K} = 0 \).
    
  \textbf{Robni pogoji}

  Togo vpeta \( \left. x \right|_{\text{rob}} = 0 \quad \text{ in } \quad  \left. \partial_z x \right|_{\text{rob}} = 0 \)

  Prost \( \left. \mathbf{M} \right|_{\text{rob}} = 0 \quad \text{ in } \quad  \left. \mathbf{F} \right|_{\text{rob}} = 0 \)

  Podprt/ vrtljivo vpet \(  \left. x \right|_{\text{rob}} = 0 \quad \text{ in } \quad \left. \mathbf{M} \right|_{\text{rob}} = 0 \)

    Sila in navor sta \( F_x (z) = - \frac{mg}{L} (z - L ) \) ter \( M_y (z) = \frac{mg}{L} (z ^2 / 12 - Lz + L ^2/ 4) \). Na steno je to \( F_x (0) = mg \) (\textbf{pozitivno!}) in \( M_y (0) = mgL /2 \).

Nova spremenljivka \( v = x^{(2)} \) in splošna rešitev je \( u(z) = A + Bz + C \cos (kz) + D \sin(kz) \), kjer je \( k ^2 = F_z / (EI) \).   Drugi člen na levi je advekcijski.

     \frac{\partial }{\partial t}G_x = - \int\limits_{S_1}^{} &\rho v_x \left( \mathbf{v}_1 \cdot \mathrm{d} \mathbf{S} \right) - \int\limits_{S_1}^{} p \, \mathrm{d} S - \int\limits_{S_2}^{} \rho v_x \left( \mathbf{v}_2 \cdot \mathrm{d} \mathbf{S} \right) \\ -& \int\limits_{S_2}^{} p \, \mathrm{d} S - \int\limits_{pl}^{} p \, \mathrm{d} S

    Velja \( \int\limits_{S_i}^{} p \mathrm{d} S = p_i S_i \) in zadnji integral je sila \( F_x \). Analogno za \( F_y \).

  \subsubsection{Cirkulacija}

  Znana vrtinčnost hitrostnega polja \( \bm{\omega} (\mathbf{r}) = \nabla \times \mathbf{v} \).

  Cirkulacija \( \Gamma = \int\limits_{}^{} \bm{\omega} \, \mathrm{d} \mathbf{S} = \oint \mathbf{v} \cdot \mathrm{d} \mathbf{r} \). 

  Kelvinov zakon pravi, da se deli vrtinčne niti gibljejo s hitrostmi, ki jih ima tekočina na mestih \( \frac{\mathrm{d} \Gamma}{\mathrm{d} t} = 0 \).

  \( \mathbf{v} (\mathbf{r}) = \frac{1}{4 \pi} = \int\limits_{}^{} \frac{\bm{\omega} (\mathbf{r}') \times (\mathbf{r} - \mathbf{r}')}{\left| \mathbf{r} - \mathbf{r}' \right|} \, \mathrm{d} ^3 \mathbf{r}' = \frac{\Gamma}{4 \pi} \int\limits_{}^{} \frac{\mathrm{d} \mathbf{l} \times (\mathbf{r} - \mathbf{r}')}{\left| \mathbf{r} - \mathbf{r}' \right| ^3} \)

  \begin{primer}
    Neskončna nit s cirkulacijo \( \Gamma \) ima hitrostno polje \( \mathbf{v}(r) = \frac{\Gamma}{2 \pi r} \hat{\mathbf{e}}_r \).
  \end{primer}
  \begin{primer}[Rankinov vrtinec]
    Hitrostno polje je

    \( v(r) = \begin{cases}\omega r &, r < a \\ \frac{\Gamma}{2 \pi r} &, r > a\end{cases} \)

    Za \( r < a \) polje ni brezvrtinčno, rešujemo z Eulerjevo enačbo za stacionarno polje, kjer velja \( \mathbf{v} \cdot \nabla \mathbf{v} = - \frac{v ^2}{r} \hat{\mathbf{e}}_{\phi} \).
  \end{primer}

  \subsubsection{2D nestisljiv brezvrtinčni tok}

  Tokovna funkcijo \( \psi \), ki je konstantna vzdolž tokovnic. Iz Cauchy-Riemmanovih enačb sledi

  \( \mathbf{v} = (v_x, v_y ) = \left( \partial_x \phi, \partial_y \phi \right) = \left( \partial_y \psi, - \partial_x \psi \right) \).

  Zapišemo kompleksni potencial \( w(z) = \phi(z) + \mathrm{i} \psi(z), \ z = x + \mathrm{i} y \) in velja \( \frac{\mathrm{d} \omega}{\mathrm{d} z} = v_x - \mathrm{i} v_y \).

  Pretok skozi poljubno krivuljo za normalo \( \hat{\mathbf{n}} = (- \mathrm{d} y, \mathrm{d} x) \) je

  \( q = \int\limits_{\mathbf{r}_1}^{\mathbf{r}_2} \left( \mathbf{v} \cdot \mathbf{n} \right) \, \mathrm{d} l = \int\limits_{\mathbf{r}_1}^{\mathbf{r}_2} \partial_l \, \mathrm{d} l = \psi(\mathbf{r}_2) - \psi(\mathbf{r}_1) \).
  
  \begin{primer}[Izvir]
    Krožnica okrog izvira poda kompleksno potencialno polje \( w(z) = \frac{q}{2 \pi} \ln z = \frac{q}{2 \pi} \left\{ \ln r + \mathrm{i} \phi \right\} \).
  \end{primer}

  Domeno pod kotom \( \alpha \) preslikamo v zgornjo polravnino, za katero vemo, da je \( w_0 = v_0 (x + \mathrm{i} y) \). Na robovih zahtevamo, da je kompleksen potencial \( w(z) = 0 \forall z \in \partial S \).

  Preslikava \( f(z) \) ohrani identiteto točkastih in neskončno oddaljenih multipolov, če slika \( 0 \to 0 \) in \( \infty \to \infty \), drugače jih zamenja, če slika \( 0 \to \infty \) in \( \infty \to 0 \). 

  \begin{primer}[Zrcaljenje preko polravnine]
    Virtualen izvir na drugi strani polravnine nam poda kompleksne potencial \( w(z) = \frac{q}{2 \pi} \left[ \ln (z - a) + \ln (z + a) \right] \). V polarni obliki je potencial \( w = \frac{Q}{2 \pi} \left( \ln r_1 + \ln r_2 \right) + \mathrm{i} \frac{Q}{2 \pi} \left( \phi_1 + \phi_2 \right) \), od koder razberemo \( \phi(r) = \mathrm{Re} w(z) \). Hitrostno polje je \( \mathbf{v} = \nabla \phi(x, y) \) in velja \( v_x = \partial_x \phi \) ter \( v_y = \partial_y \phi \). Sila na dolžinsko enoto na steno je \( F_y / l = - \int\limits_{- \infty}^{\infty} p (0, y) \, \mathrm{d} l \), kjer tlak izračunamo s pomočjo Bernoullijeve enačbe \( p + \frac{1}{2} \rho v ^2 = \mathrm{konst} = 0 \). Eni komponenti hitrostnega polja se seštejeta, drugi se odštejeta. V primeru 3D točkastega izvora je hitrostno polje \( v(r) = Q / (4 \pi r ^2) \), ter sila je integral po površini \( F_y / S = - \int\limits_0^{\infty} \int\limits_0^{2 \pi} p(r, a) r\, \mathrm{d} r \).
  \end{primer}

  \begin{primer}
    Vrtinčna nit ima kompleksen potencial \( w_{\Gamma} (z) = - \mathrm{i} \frac{\Gamma}{2 \pi} \ln z \). 
  \end{primer}

  \subsubsection{Viskozne tekočine}  

  Navier- Stokes \( \rho \left[ \partial_t \mathbf{v} + \left( \mathbf{v} \cdot \nabla \right) \mathbf{v} \right] = - \nabla p + \nabla \cdot \sigma_{ij}^v + \mathbf{f} \)

  Viskozni napetostni tenzor \( \sigma_{ij}^{v} = 2 \eta \left( v_{ij} - \frac{1}{3} v_{kk} \delta_{ij} \right) + \zeta v_{kk} \delta_{ij} \) in gradient hitrosti \( v_{ij} = \frac{1}{2} \left( \partial_j v_i + \partial_i v_j \right) \). \( \eta \) viskoznost, \( \zeta \) dilatacijska viskoznost.

  Navier-Stokes običajna oblika \( \rho \left( \partial_t \mathbf{v} + \left( \mathbf{v} \cdot \nabla \right) \mathbf{v} \right) = - p + \eta \nabla ^2 \mathbf{v} + \left( \zeta + \frac{1}{3} \eta \right) \nabla \nabla \cdot \mathbf{v} + \mathbf{f} \)

  \( \mathbf{f} = 0 \) in \( \rho = 0 \) za nestisljive tekočine je Helmholtzova enačba za vrtinčnost \( \bm{\omega} = \nabla \times \mathbf{v} \)

  \( \partial_t \bm{\omega} + \left( \mathbf{v} \cdot \nabla \right) \bm{\omega} = \left( \bm{\omega} \cdot \nabla \right) \mathbf{v} + \frac{\eta}{\rho} \nabla ^2 \bm{\omega} \).

  \begin{primer}
    Plošča pri \( z = 0 \) se začne premikati po \( t > 0 \) proti \( + x \) z \( v_0 \). Vrtinčnost \( \bm{\omega} = \omega \hat{\mathbf{e}}_y \). Helmholtz v difuzijsko enačbo, ki ima vključen začetni pogoj

    \( \partial_t \bm{\omega} + D \nabla ^2 \bm{\omega} = v_0 \delta(z) \delta(t)  \).

    Rešitev je Greenova funcija, ki jo preslikamo še na območje \( z < 0 \), saj splošna rešitev velja z \( - \infty < z < \infty \).

    \( \omega(z) = \frac{1}{\sqrt{4 \pi D t}} \exp \left\{ - \frac{z ^2}{4 D t} \right\} \).

    Rešitev je potem

    \( \omega(z, t) = - 2 v_0 \frac{1}{\sqrt{4 \pi D t}} \exp \left\{ - \frac{z ^2}{4 D t} \right\} \).
  \end{primer}

  \begin{primer}
    Hitrostno polje s cirkulacijo \( \mathbf{v}(r) = \frac{\Gamma}{2 \pi r} \hat{\mathbf{e}}_r \). Vrtinčnost je \( \bm{\omega} = \Gamma \delta ^2(\mathbf{r})  \hat{\hat{e}}_z \). Rešujemo difuzijsko enačbo

    \( \partial_t u - D \left( \partial_x ^2 u + \partial_y ^2 y \right) = \delta(x) \delta(y) \delta(t) \).

    Rešitev je združena Greenova funkcija za \( x \) in \( y \)

    \( \omega(\mathbf{r}, t) = \frac{\Gamma}{4 \pi D t} \exp \left\{ - \frac{r ^2}{4 \pi D t} \right\} \). 

    Hitrostno polje je preko Stokesovega izreka

    \( \oint_{\partial S} \mathbf{v} \cdot \mathrm{d} \mathbf{l} = \int\limits_S^{} \bm{\omega} \cdot \mathrm{d} \mathbf{S} \)

    potem

    \( v \cdot 2 \pi r = 2 \pi \int\limits_0^r \omega r \cdot \mathrm{d} r = \Gamma \left( 1 - \exp \left\{ - \frac{r ^2}{4 \pi D t} \right\} \right) \).
    \end{primer}
