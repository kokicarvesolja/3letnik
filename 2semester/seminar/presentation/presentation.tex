\begin{frame}
  \frametitle{Nastanek mehurčka -~toplotni model}
  \begin{block}{Nastanek stabilnega mehurček}
    \begin{itemize}
      \item <1-> energija mehurčka \( E_m \)
      \item <2-> razlika tlakov $ p_g - p_f = 2 \sigma / R  $
      \item <3-> mehurček kritične velikosti s polmerom \( R_C = 2 \sigma / p_g \)
    \end{itemize}
    \vspace{1cm}
    \only<1-2>{
    \[ E_m = 4 \pi \sigma R ^2 + \frac{4\pi}{3} R ^3 n H_s
    \]}
    \only<3->{
    \[ E_m \approx 16 \pi \frac{\sigma ^3}{p_g ^2} \left( 1 + \frac{2}{3} \frac{n}{p_g} H_s \right) 
    \]
    }
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Nastanek mehurčka -~toplotni model}
  \begin{block}{Sipanje vpadnega delca na tekočini}
    \begin{itemize}
      \item <1-> sipalni presek \( \sigma_{e, m} \) za prenos energije večje od energijskega pragu \( E_t \) na molekulo
      \item <2-> Seitz oceni \( \sigma_{e,m} \) iz gostote mehurčkov na sledi vpadnega delca
      \item <3-> sipalni presek za prenos energije pri Rutherfordovem sipanju na elektronu
      \item <4-> energijski prag \( E_t \) na molekulo
    \end{itemize}

  \end{block}
  \only<3>{
    \[ \sigma_e = 8 \pi r_b ^2  \frac{R_h ^2}{m_e v ^2 E_t} 
    \]}
  \only<4->{
    \[   E_t = 8 \pi r_b ^2 \frac{R_h ^2 Z_e}{m_e v ^2 \sigma_{e, m}}
    \]}
\end{frame}

\begin{frame}
  \frametitle{Nastanek mehurčka -~toplotni model}
  \begin{block}{Hitrost sten mehurčka}
    \begin{itemize}
      \item<1-> žarek \( \delta \) odda energijo v kroglo s polmerom \( R_0 < R_C \). 
      \item<2-> difuzija toplote \( \tau_h = r ^2 / (4D) \)
      \item <3-> hitrost sten mehurčka
    \end{itemize}

    \only<3->{
      \[ v_C = \frac{R_C}{\tau_C} = \frac{4D}{R_C}. 
      \]
    }
  \end{block}

  \only<4>{
    \begin{block}{Hitrost zvoka v snovi}
      \[ c_a = \left( \rho \beta_a \right)^{-1/2} 
      \]
    \end{block}}
\end{frame}

\begin{frame}
\frametitle{Razširjanje mehurčka -~toplotni model}
\begin{itemize}
  \item <1-> radialno širjenje mehurčka z začetnim polmerom \( R_0 \)
  \item <2-> nestisljiva in neviskozna tekočina
  \item <3-> Eulerjeva enačba in kontinuitetna enačba
\end{itemize}

\only<3->{
  \[ \frac{\partial v}{\partial t} + v \frac{\partial v}{\partial r} = - \frac{1}{\rho} \frac{\partial p}{\partial r} 
  \]
  \[ \frac{\partial }{\partial r}  \left( r ^2 v \right) = 0 
  \]
}

\only<4->{
  Uporabimo nastavek \( v = \dot{R} \frac{R ^2}{r ^2} \).
}
\end{frame}

\begin{frame}
\frametitle{Razširjanje mehurčka -~toplotni model}
\begin{block}{Fundamentalna enačba dinamike mehurčkov}
  \[ \frac{p}{\rho} = \ddot{R} R + \frac{3}{2} {\left( \dot{R} \right)}^2
  \]

  \begin{itemize}
    \item adiabatna sprememba
    \item vodik z \( \gamma = 4 / 3 \)
  \end{itemize}
\end{block}

\vspace{1cm}

\only<2->{
  Kaj pa viskoznost tekočine?
}
\end{frame}

\begin{frame}
\frametitle{Razširjanje mehurčka -~toplotni model}
\begin{block}{Viskozni člen}
  \begin{itemize}
    \item <1-> upočasnjuje rast mehurčka
    \item <2-> enakost tlakov \( p_R = p \)
    \item <3-> razmerje med mehurčkovim in kritičnim polmerom
  \end{itemize}

\end{block}
\only<1-2>{
  \[ p_R = - 4 \eta \frac{\dot{R}}{R} 
  \]
}
\only<3->{
\[  {\left( \frac{R}{R_C} \right)}^{3\gamma - 2} = \left( \frac{9}{128 \pi} \right) \left( \frac{\rho}{\eta ^2} \right) \left( \frac{p_0 V_0}{R_C} \right) {\left( \frac{R_0}{R_C} \right)}^{3(\gamma - 1)}.
\]
}
\end{frame}

\end{itemize}
\end{block}

\end{frame}

\begin{frame}
\frametitle{Razširjanje mehurčka -~toplotni model}
\begin{block}{Širjenje preko kritične velikosti}
  \begin{itemize}
    \item<1-> izhlapevanje tekočine
    \item <2-> temperaturni gradient \( T_b - T_v \) debeline \( 2 \left( D t \right)^{1/2} \)
    \item <3-> asimptotska rast mehurčka
  \end{itemize}
\end{block}
\only<3->{
\[ R \approx 2 \lambda \left( T_v - T_b \right) \frac{t^{1/2}}{\rho' H_s D^{1/2}} 
\]}
\end{frame}
